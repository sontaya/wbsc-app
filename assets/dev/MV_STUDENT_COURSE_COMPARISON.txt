## 📝 SQL ที่แก้ไขตามชื่อใหม่

```sql
-- ✅ สร้าง Materialized View ชื่อใหม่
CREATE MATERIALIZED VIEW WBSC.MV_STUDENT_COURSE_COMPARISON
BUILD IMMEDIATE
REFRESH COMPLETE ON DEMAND
AS
SELECT
    ENR.STD_CODE,
    ENR.STD_ENROLL_KEY,
    ENR.USERNAME,
    ENR.FIRST_NAME_TH || ' ' || ENR.LAST_NAME_TH AS FULLNAME,
    ENR.FACULTYCODE,
    ENR.FACULTYNAMETH,
    ENR.SUB_NAME_TH,
    ENR.SUB_SHOW,
    ENR.SECTION_CODE,
    ENR.STATUS_NAME_TH,
    ENR.IS_APPROVE,
    ENR.SUB_COUNT,
    ENR.STATE_DIRECTION,  -- เก็บค่านี้ไว้ใน MV
    ENR.ACADEMIC_YEAR,
    ENR.SEMESTER,
    SD.CAT_TERM_ID,
    SD.CAT_TERM,
    SD.STUDENT_GROUPS,
    CASE
        WHEN SD.STD_ENROLL_KEY IS NULL THEN 'MISSING_IN_MOODLE'
        ELSE 'EXISTS_IN_MOODLE'
    END AS SYNC_STATUS,
    -- ⭐ เพิ่ม column ช่วยแยกแยะ
    CASE
        WHEN ENR.STATE_DIRECTION = 'KEEP COURSE'
             AND SD.STD_ENROLL_KEY IS NULL
        THEN 'NEED_TO_ADD'
        WHEN ENR.STATE_DIRECTION = 'DROP COURSE'
             AND SD.STD_ENROLL_KEY IS NOT NULL
        THEN 'NEED_TO_REMOVE'
        WHEN ENR.STATE_DIRECTION = 'KEEP COURSE'
             AND SD.STD_ENROLL_KEY IS NOT NULL
        THEN 'ALREADY_SYNCED'
        ELSE 'OTHER'
    END AS ACTION_REQUIRED
FROM WBSC.VW_WBSC_ENROLL682 ENR
LEFT JOIN WBSC.COURSE_STUDENT_DETAILED SD
    ON ENR.STD_ENROLL_KEY = SD.STD_ENROLL_KEY
WHERE ENR.IS_APPROVE = '1';  -- เก็บเฉพาะที่อนุมัติแล้ว;

-- สร้าง Index
CREATE INDEX idx_mv_scc_faculty_term
ON WBSC.MV_STUDENT_COURSE_COMPARISON(FACULTYCODE, CAT_TERM_ID);

CREATE INDEX idx_mv_scc_sync_status
ON WBSC.MV_STUDENT_COURSE_COMPARISON(SYNC_STATUS, CAT_TERM_ID);

CREATE INDEX idx_mv_scc_enroll_key
ON WBSC.MV_STUDENT_COURSE_COMPARISON(STD_ENROLL_KEY);

-- Query
SELECT *
FROM WBSC.MV_STUDENT_COURSE_COMPARISON
WHERE CAT_TERM_ID = 273
ORDER BY FACULTYCODE, SUB_SHOW, STD_CODE;

-- Refresh MV (Manual)
EXEC DBMS_MVIEW.REFRESH('WBSC.MV_STUDENT_COURSE_COMPARISON');

-- ตรวจสอบสถานะ
SELECT
    MVIEW_NAME,
    LAST_REFRESH_DATE,
    STALENESS,
    COMPILE_STATE
FROM USER_MVIEWS
WHERE MVIEW_NAME = 'MV_STUDENT_COURSE_COMPARISON';
```

-- สร้าง Job Refresh MV ทุก 30 นาที
BEGIN
    DBMS_SCHEDULER.CREATE_JOB (
        job_name        => 'JOB_REFRESH_STUDENT_COURSE_MV',
        job_type        => 'PLSQL_BLOCK',
        job_action      => 'BEGIN DBMS_MVIEW.REFRESH(''WBSC.MV_STUDENT_COURSE_COMPARISON'', ''C''); END;',
        start_date      => SYSDATE,
        repeat_interval => 'FREQ=MINUTELY; INTERVAL=30',
        enabled         => TRUE,
        comments        => 'Refresh Student Course Comparison MV every 30 minutes'
    );
END;
/

-- ตรวจสอบ Job
SELECT
    JOB_NAME,
    STATE,

    NEXT_RUN_DATE,
    RUN_COUNT

WHERE JOB_NAME = 'JOB_REFRESH_STUDENT_COURSE_MV';

BEGIN
    DBMS_SCHEDULER.DROP_JOB('JOB_REFRESH_STUDENT_COURSE_MV');
END;
/
