# ระบบบริหารจัดการการเพิ่มถอนรายวิชา Moodle LMS

## 📋 ภาพรวมระบบ

ระบบนี้ทำหน้าที่เชื่อมต่อระหว่าง Oracle Database (ระบบทะเบียน) กับ Moodle LMS โดยอัตโนมัติ เพื่อจัดการการเพิ่ม/ถอนรายวิชาของนักศึกษาให้ตรงกับข้อมูลการลงทะเบียนจริง

### เทคโนโลยีที่ใช้
- **Backend**: CodeIgniter 4 + PHP 8.3
- **Database**: MariaDB (Moodle) + Oracle 11g (Source)
- **Server**: Apache + Docker
- **Moodle Plugin**: Flat file (CSV) enrolment

## 🔄 Data Flow & Process

### 1. Sync Process Flow

```
START
  │
  ├─→ [1] เชื่อมต่อ Oracle DB
  │        │
  │        ├─→ ดึงข้อมูลจาก View (ภาคการศึกษาปัจจุบัน)
  │        └─→ จัดเก็บข้อมูลชั่วคราว
  │
  ├─→ [2] เชื่อมต่อ Moodle DB
  │        │
  │        └─→ ดึงข้อมูล Master Data ที่มีอยู่
  │
  ├─→ [3] เปรียบเทียบข้อมูล
  │        │
  │        ├─→ ระบุรายการที่ต้อง ADD (มีใน Oracle ไม่มีใน Moodle)
  │        ├─→ ระบุรายการที่ต้อง DEL (มีใน Moodle ไม่มีใน Oracle)
  │        └─→ ระบุรายการที่ไม่เปลี่ยนแปลง
  │
  ├─→ [4] อัพเดท Master Data
  │        │
  │        ├─→ INSERT รายการใหม่ (status = active)
  │        ├─→ UPDATE รายการเดิม (status = inactive)
  │        └─→ บันทึก Log
  │
  ├─→ [5] สร้างไฟล์ CSV
  │        │
  │        ├─→ Format: operation,role,user_idnumber,course_idnumber
  │        ├─→ บันทึกไฟล์: /VolFreenas/.../sync-data.csv
  │        └─→ Validate ไฟล์
  │
  ├─→ [6] Execute Moodle Task
  │        │
  │        └─→ php74 scheduled_task.php --execute='\enrol_flatfile\task\flatfile_sync_task'
  │
  ├─→ [7] บันทึกผลลัพธ์
  │        │
  │        ├─→ อัพเดท Sync Log
  │        ├─→ บันทึก Errors (ถ้ามี)
  │        └─→ ส่งอีเมลแจ้งเตือน (ถ้าเปิดใช้)
  │
END
```

### 2. Business Logic

```php
// Pseudo Code for Sync Logic

foreach (Oracle Records as $oracleRecord) {
    $key = $oracleRecord->user_idnumber . '_' . $oracleRecord->course_idnumber;
    $masterRecord = MasterData[$key] ?? null;

    if (!$masterRecord) {
        // กรณีที่ 1: มีใน Oracle แต่ไม่มีใน Master
        → INSERT Master Data (status = 'active')
        → Add to CSV (operation = 'add')

    } elseif ($masterRecord->status == 'inactive') {
        // กรณีที่ 2: เคยถอนแล้ว แต่ลงทะเบียนใหม่
        → UPDATE Master Data (status = 'active')
        → Add to CSV (operation = 'add')

    } else {
        // กรณีที่ 3: ไม่มีการเปลี่ยนแปลง
        → UPDATE last_sync_date only
        → Skip CSV
    }
}

// ตรวจสอบรายการที่ถูกถอน
foreach (MasterData as $masterRecord) {
    if ($masterRecord->status == 'active') {
        $key = $masterRecord->user_idnumber . '_' . $masterRecord->course_idnumber;

        if (!OracleRecords[$key]) {
            // กรณีที่ 4: มีใน Master แต่ไม่มีใน Oracle
            → UPDATE Master Data (status = 'inactive')
            → Add to CSV (operation = 'del')
        }
    }
}
```

### 2. Libraries/OracleService.php

```php
<?php
namespace App\Libraries;

use Config\Database;

class OracleService
{
    protected $db;
    protected $viewName;

    public function __construct()
    {
        $this->db = Database::connect('oracle');
        $this->viewName = env('oracle.viewName', 'V_STUDENT_ENROLLMENT');
    }

    /**
     * ดึงข้อมูลการลงทะเบียนจาก Oracle View
     */
    public function getEnrollmentData(string $semester): array
    {
        $sql = "
            SELECT
                STUDENT_ID as user_idnumber,
                COURSE_CODE as course_idnumber,
                SEMESTER as semester,
                STUDENT_NAME as student_name,
                COURSE_NAME as course_name,
                SECTION as section,
                CREDIT as credit
            FROM {$this->viewName}
            WHERE SEMESTER = :semester
            AND STATUS = 'E'
            ORDER BY STUDENT_ID, COURSE_CODE
        ";

        $query = $this->db->query($sql, ['semester' => $semester]);
        return $query->getResultArray();
    }

    /**
     * ตรวจสอบการเชื่อมต่อ
     */
    public function testConnection(): bool
    {
        try {
            $query = $this->db->query("SELECT 1 FROM DUAL");
            return $query !== false;
        } catch (\Exception $e) {
            log_message('error', 'Oracle connection failed: ' . $e->getMessage());
            return false;
        }
    }
}
```

### 3. Libraries/MoodleEnrolService.php

```php
<?php
namespace App\Libraries;

use App\Models\EnrolOracleSyncModel;
use App\Models\EnrolOracleSyncLogModel;
use App\Models\EnrolOracleSyncErrorModel;

class MoodleEnrolService
{
    protected $syncModel;
    protected $logModel;
    protected $errorModel;
    protected $csvGenerator;
    protected $oracleService;

    public function __construct()
    {
        $this->syncModel = new EnrolOracleSyncModel();
        $this->logModel = new EnrolOracleSyncLogModel();
        $this->errorModel = new EnrolOracleSyncErrorModel();
        $this->csvGenerator = new \App\Libraries\CsvGenerator();
        $this->oracleService = new \App\Libraries\OracleService();
    }

    /**
     * ฟังก์ชันหลักสำหรับ Sync ข้อมูล
     */
    public function syncEnrollments(string $semester, string $triggeredBy = 'system'): array
    {
        $logId = $this->createSyncLog($semester, $triggeredBy);
        $startTime = microtime(true);

        try {
            // 1. ดึงข้อมูลจาก Oracle
            $oracleData = $this->oracleService->getEnrollmentData($semester);

            if (empty($oracleData)) {
                throw new \Exception('No data from Oracle for semester: ' . $semester);
            }

            // 2. ดึงข้อมูล Master Data
            $masterData = $this->syncModel->where('semester', $semester)->findAll();
            $masterDataMap = $this->buildDataMap($masterData);

            // 3. เปรียบเทียบและจัดเตรียมข้อมูล
            $syncResult = $this->compareAndPrepareData(
                $oracleData,
                $masterDataMap,
                $semester
            );

            // 4. อัพเดท Master Data
            $this->updateMasterData($syncResult);

            // 5. สร้างไฟล์ CSV
            $csvPath = $this->csvGenerator->generate(
                $syncResult['csvData'],
                $semester
            );

            // 6. Execute Moodle Flat File Task
            $this->executeMoodleTask();

            // 7. อัพเดท Log
            $executionTime = microtime(true) - $startTime;
            $this->updateSyncLog($logId, $syncResult, $csvPath, $executionTime);

            return [
                'success' => true,
                'log_id' => $logId,
                'stats' => $syncResult['stats'],
                'execution_time' => $executionTime
            ];

        } catch (\Exception $e) {
            $this->logModel->update($logId, [
                'status' => 'failed',
                'error_message' => $e->getMessage()
            ]);

            log_message('error', 'Sync failed: ' . $e->getMessage());

            return [
                'success' => false,
                'log_id' => $logId,
                'error' => $e->getMessage()
            ];
        }
    }

    /**
     * เปรียบเทียบข้อมูลและจัดเตรียม
     */
    protected function compareAndPrepareData(array $oracleData, array $masterMap, string $semester): array
    {
        $csvData = [];
        $toInsert = [];
        $toUpdate = [];
        $stats = [
            'total' => count($oracleData),
            'new' => 0,
            'reactivate' => 0,
            'unchanged' => 0,
            'removed' => 0
        ];

        $now = date('Y-m-d H:i:s');

        // ตรวจสอบรายการใน Oracle
        foreach ($oracleData as $record) {
            $key = $record['user_idnumber'] . '_' . $record['course_idnumber'];

            if (!isset($masterMap[$key])) {
                // กรณีที่ 1: รายการใหม่
                $toInsert[] = array_merge($record, [
                    'status' => 'active',
                    'first_sync_date' => $now,
                    'last_sync_date' => $now,
                    'last_action' => 'add',
                    'sync_count' => 1
                ]);

                $csvData[] = [
                    'operation' => 'add',
                    'role' => 'student',
                    'user_idnumber' => $record['user_idnumber'],
                    'course_idnumber' => $record['course_idnumber']
                ];

                $stats['new']++;

            } elseif ($masterMap[$key]['status'] === 'inactive') {
                // กรณีที่ 2: Reactivate
                $toUpdate[] = [
                    'id' => $masterMap[$key]['id'],
                    'status' => 'active',
                    'last_sync_date' => $now,
                    'last_action' => 'add',
                    'sync_count' => $masterMap[$key]['sync_count'] + 1
                ];

                $csvData[] = [
                    'operation' => 'add',
                    'role' => 'student',
                    'user_idnumber' => $record['user_idnumber'],
                    'course_idnumber' => $record['course_idnumber']
                ];

                $stats['reactivate']++;

            } else {
                // กรณีที่ 3: ไม่เปลี่ยนแปลง
                $toUpdate[] = [
                    'id' => $masterMap[$key]['id'],
                    'last_sync_date' => $now,
                    'sync_count' => $masterMap[$key]['sync_count'] + 1
                ];

                $stats['unchanged']++;
            }

            unset($masterMap[$key]); // ลบออกจาก map
        }

        // ตรวจสอบรายการที่เหลือใน Master (ถูกถอน)
        foreach ($masterMap as $key => $record) {
            if ($record['status'] === 'active') {
                $toUpdate[] = [
                    'id' => $record['id'],
                    'status' => 'inactive',
                    'last_sync_date' => $now,
                    'last_action' => 'del'
                ];

                $csvData[] = [
                    'operation' => 'del',
                    'role' => 'student',
                    'user_idnumber' => $record['user_idnumber'],
                    'course_idnumber' => $record['course_idnumber']
                ];

                $stats['removed']++;
            }
        }

        return [
            'toInsert' => $toInsert,
            'toUpdate' => $toUpdate,
            'csvData' => $csvData,
            'stats' => $stats
        ];
    }

    /**
     * อัพเดท Master Data
     */
    protected function updateMasterData(array $syncResult): void
    {
        $db = \Config\Database::connect();
        $db->transStart();

        try {
            // Insert ข้อมูลใหม่
            if (!empty($syncResult['toInsert'])) {
                $this->syncModel->insertBatch($syncResult['toInsert']);
            }

            // Update ข้อมูลเดิม
            if (!empty($syncResult['toUpdate'])) {
                $this->syncModel->updateBatch($syncResult['toUpdate'], 'id');
            }

            $db->transComplete();

            if ($db->transStatus() === false) {
                throw new \Exception('Transaction failed');
            }

        } catch (\Exception $e) {
            $db->transRollback();
            throw $e;
        }
    }

    /**
     * Execute Moodle Flat File Task
     */
    protected function executeMoodleTask(): bool
    {
        $command = "php74 /path/to/moodle/admin/cli/scheduled_task.php --execute='\enrol_flatfile\task\flatfile_sync_task' 2>&1";

        exec($command, $output, $returnVar);

        log_message('info', 'Moodle task executed: ' . implode("\n", $output));

        return $returnVar === 0;
    }

    /**
     * สร้าง Data Map
     */
    protected function buildDataMap(array $data): array
    {
        $map = [];
        foreach ($data as $record) {
            $key = $record['user_idnumber'] . '_' . $record['course_idnumber'];
            $map[$key] = $record;
        }
        return $map;
    }

    /**
     * สร้าง Sync Log
     */
    protected function createSyncLog(string $semester, string $triggeredBy): int
    {
        return $this->logModel->insert([
            'sync_datetime' => date('Y-m-d H:i:s'),
            'sync_type' => ($triggeredBy === 'system') ? 'scheduled' : 'manual',
            'semester' => $semester,
            'status' => 'running',
            'triggered_by' => $triggeredBy
        ]);
    }

    /**
     * อัพเดท Sync Log
     */
    protected function updateSyncLog(int $logId, array $syncResult, string $csvPath, float $executionTime): void
    {
        $this->logModel->update($logId, [
            'total_records' => $syncResult['stats']['total'],
            'new_enrolls' => $syncResult['stats']['new'] + $syncResult['stats']['reactivate'],
            'removed_enrolls' => $syncResult['stats']['removed'],
            'unchanged' => $syncResult['stats']['unchanged'],
            'execution_time' => round($executionTime, 2),
            'csv_file_path' => $csvPath,
            'csv_file_size' => file_exists($csvPath) ? filesize($csvPath) : 0,
            'status' => 'completed'
        ]);
    }
}
```

### 4. Libraries/CsvGenerator.php

```php
<?php
namespace App\Libraries;

class CsvGenerator
{
    protected $csvPath;

    public function __construct()
    {
        $this->csvPath = env('csv.filePath', '/VolFreenas/wbsc2021/moodledata/enroll_files/sync-data.csv');
    }

    /**
     * สร้างไฟล์ CSV
     */
    public function generate(array $data, string $semester): string
    {
        // สร้าง backup ของไฟล์เดิม
        if (file_exists($this->csvPath)) {
            $backupPath = str_replace('.csv', '_' . date('YmdHis') . '.csv', $this->csvPath);
            copy($this->csvPath, $backupPath);
        }

        // สร้างไฟล์ใหม่
        $fp = fopen($this->csvPath, 'w');

        if (!$fp) {
            throw new \Exception('Cannot create CSV file: ' . $this->csvPath);
        }

        // เขียน Header (ถ้าต้องการ - แต่ Moodle flat file ไม่ต้องการ header)
        // fputcsv($fp, ['operation', 'role', 'user', 'course']);

        // เขียนข้อมูล
        foreach ($data as $row) {
            fputcsv($fp, [
                $row['operation'],
                $row['role'],
                $row['user_idnumber'],
                $row['course_idnumber']
            ]);
        }

        fclose($fp);

        // Validate ไฟล์
        if (!file_exists($this->csvPath)) {
            throw new \Exception('CSV file not created');
        }

        log_message('info', 'CSV generated: ' . $this->csvPath . ' (' . count($data) . ' records)');

        return $this->csvPath;
    }

    /**
     * อ่านไฟล์ CSV
     */
    public function read(): array
    {
        if (!file_exists($this->csvPath)) {
            return [];
        }

        $data = [];
        $fp = fopen($this->csvPath, 'r');

        while (($row = fgetcsv($fp)) !== false) {
            $data[] = [
                'operation' => $row[0],
                'role' => $row[1],
                'user_idnumber' => $row[2],
                'course_idnumber' => $row[3]
            ];
        }

        fclose($fp);
        return $data;
    }
}
```

```php
<?php
namespace App\Controllers\Admin;

use App\Controllers\BaseController;
use App\Libraries\MoodleEnrolService;
use App\Models\EnrolOracleSyncLogModel;
use App\Models\EnrolOracleSyncConfigModel;

class SyncController extends BaseController

    protected $logModel;
    protected $configModel;

    public function __construct()
    {

        $this->logModel = new EnrolOracleSyncLogModel();
        $this->configModel = new EnrolOracleSyncConfigModel();
    }

    /**
     * หน้าแรก - Dashboard
     */
    public function index()
    {
        $data = [
            'title' => 'Moodle Enrollment Sync Manager',
            'currentSemester' => $this->configModel->getValue('current_semester'),
            'lastSync' => $this->logModel->orderBy('sync_datetime', 'DESC')->first(),
            'recentLogs' => $this->logModel->orderBy('sync_datetime', 'DESC')->limit(10)->findAll(),
            'stats' => $this->getStats()
        ];

        return view('admin/sync/index', $data);
    }

    /**
     * Manual Sync
     */
    public function manualSync()
    {
        if (!$this->request->isAJAX()) {
            return redirect()->back();
        }

        $semester = $this->request->getPost('semester')
                    ?? $this->configModel->getValue('current_semester');

        $result = $this->moodleService->syncEnrollments(
            $semester,
            session()->get('username') ?? 'admin'
        );

        return $this->response->setJSON($result);
    }

    /**
     * ดูประวัติการ Sync
     */
    public function history()
    {
        $data = [
            'title' => 'Sync History',
            'logs' => $this->logModel
                ->orderBy('sync_datetime', 'DESC')
                ->paginate(20),
            'pager' => $this->logModel->pager
        ];

        return view('admin/sync/history', $data);
    }

    /**
     * ดูรายละเอียด Log
     */
    public function viewLog($logId)
    {
        $log = $this->logModel->find($logId);

        if (!$log) {
            return redirect()->to('/admin/sync/history')
                ->with('error', 'Log not found');
        }

        $data = [
            'title' => 'Sync Log Details',
            'log' => $log,
            'errors' => $this->logModel->getErrors($logId)
        ];

        return view('admin/sync/log_detail', $data);
    }

    /**
     * ดาวน์โหลด CSV
     */
    public function downloadCsv($logId)
    {
        $log = $this->logModel->find($logId);

        if (!$log || !file_exists($log['csv_file_path'])) {
            return redirect()->back()->with('error', 'CSV file not found');
        }

        return $this->response->download($log['csv_file_path'], null);
    }

    /**
     * สถิติ
     */
    protected function getStats(): array
    {
        $db = \Config\Database::connect();

        return [
            'total_active' => $db->table('mdl_enrol_oracle_sync')
                ->where('status', 'active')
                ->countAllResults(),
            'total_inactive' => $db->table('mdl_enrol_oracle_sync')
                ->where('status', 'inactive')
                ->countAllResults(),
            'today_syncs' => $db->table('mdl_enrol_oracle_sync_log')
                ->where('DATE(sync_datetime)', date('Y-m-d'))
                ->countAllResults(),
            'failed_syncs' => $db->table('mdl_enrol_oracle_sync_log')
                ->where('status', 'failed')
                ->where('DATE(sync_datetime)', date('Y-m-d'))
                ->countAllResults()
        ];
    }
}
```

```php
    }
}
```

```javascript
document.addEventListener('DOMContentLoaded', function() {
    const form = document.getElementById('manualSyncForm');
    const progress = document.getElementById('syncProgress');
    const result = document.getElementById('syncResult');
    const syncBtn = document.getElementById('syncBtn');

    form.addEventListener('submit', async function(e) {
        e.preventDefault();

        // Disable button
        syncBtn.disabled = true;
        syncBtn.textContent = 'Syncing...';

        // Show progress
        progress.style.display = 'block';
        result.style.display = 'none';

        const formData = new FormData(form);

        try {
            const response = await fetch('/admin/sync/manual', {
                method: 'POST',
                body: formData
            });

            const data = await response.json();

            // Hide progress
            progress.style.display = 'none';
            result.style.display = 'block';

            if (data.success) {
                result.className = 'alert alert-success';
                result.innerHTML = `
                    <h4>✅ Sync Completed Successfully!</h4>
                    <ul>
                        <li>Total Records: ${data.stats.total}</li>
                        <li>New Enrolls: ${data.stats.new}</li>
                        <li>Reactivated: ${data.stats.reactivate}</li>
                        <li>Removed: ${data.stats.removed}</li>
                        <li>Unchanged: ${data.stats.unchanged}</li>
                        <li>Execution Time: ${data.execution_time.toFixed(2)}s</li>
                    </ul>
                    <a href="/admin/sync/log/${data.log_id}" class="btn-link">View Details</a>
                `;

                // Reload page after 3 seconds
                setTimeout(() => location.reload(), 3000);
            } else {
                result.className = 'alert alert-error';
                result.innerHTML = `
                    <h4>❌ Sync Failed</h4>
                    <p>${data.error}</p>
                `;
            }

        } catch (error) {
            progress.style.display = 'none';
            result.style.display = 'block';
            result.className = 'alert alert-error';
            result.innerHTML = `
                <h4>❌ Error</h4>
                <p>${error.message}</p>
            `;
        } finally {
            syncBtn.disabled = false;
            syncBtn.textContent = 'Start Sync';
        }
    });
```

1. **Database Security**
   - ใช้ Prepared Statements ทุกครั้ง
   - แยก User สำหรับ Read/Write
   - Encrypt sensitive config

2. **File Security**
   - ตรวจสอบ File permissions (CSV file)
   - Validate file path
   - Backup ไฟล์เก่าก่อนเขียนใหม่

3. **Authentication & Authorization**
   - ใช้ Middleware สำหรับ Admin Panel
   - API Key สำหรับ API Endpoints
   - Session management

4. **Logging**
   - Log ทุก action
   - Store sensitive data เฉพาะที่จำเป็น
   - Rotate logs ตามกำหนด
